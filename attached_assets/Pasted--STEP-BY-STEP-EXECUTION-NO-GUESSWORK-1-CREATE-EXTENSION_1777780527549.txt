3️⃣ CONTENT SCRIPT (CORE LOGIC)

This is the most important part.

// content.js

function detectTextInputs() {
  return document.querySelectorAll("textarea, input[type='text']");
}

async function analyzeText(text) {
  const response = await fetch("https://aurora-shield--PurvanshBhatt.replit.app/api/analyze", {
    method: "POST",
    headers: {
      "Content-Type": "application/json"
    },
    body: JSON.stringify({
      type: "prompt",
      data: text
    })
  });

  return await response.json();
}

function showWarning(element, risk) {
  let banner = document.createElement("div");
  banner.innerText = `⚠️ Risk Score: ${risk}`;
  banner.style = `
    position:absolute;
    background:red;
    color:white;
    padding:5px;
    z-index:9999;
  `;
  element.parentNode.appendChild(banner);
}

function monitorInputs() {
  const inputs = detectTextInputs();

  inputs.forEach(input => {
    input.addEventListener("blur", async () => {
      const text = input.value;
      if (!text) return;

      const result = await analyzeText(text);

      if (result.risk_score > 60) {
        showWarning(input, result.risk_score);
      }
    });
  });
}

monitorInputs();
